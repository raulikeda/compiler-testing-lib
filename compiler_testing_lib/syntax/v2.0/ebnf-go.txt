PROGRAM = { STATEMENT } ;
STATEMENT = (IDENTIFIER, "=", EXPRESSION | "Println", "(", EXPRESSION, ")" | ε), "\n" ;
EXPRESSION = TERM, { ("+" | "-"), TERM } ;
TERM = FACTOR, { ("*" | "/"), FACTOR } ;
FACTOR = NUMBER | IDENTIFIER | ("+" | "-"), FACTOR | "(", EXPRESSION, ")" ;
NUMBER = DIGIT, {DIGIT} ;
IDENTIFIER = LETTER, {LETTER | DIGIT | "_"} ;
DIGIT = "0" | "1" | ... | "9" ;
LETTER = "a" | "b" | ... | "z" | "A" | "B" | ... | "Z" ;
